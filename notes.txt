2. ProductController.cs
Replace the entire file with:
using Microsoft.AspNetCore.Mvc;
using ShoppingCart.Models;

namespace ShoppingCart.Controllers;

public class ProductController : Controller
{
    static List<Product> products = new List<Product>
    {
        new Product { Id = 1, Name = "Mouse", Price = 500 },
        new Product { Id = 2, Name = "Keyboard", Price = 1000 },
        new Product { Id = 3, Name = "Headphones", Price = 1500 }
    };

    // READ
    public IActionResult Index()
    {
        return View(products);
    }

    // UPDATE - Show edit form
    public IActionResult Edit(int id)
    {
        var product = products.Find(p => p.Id == id);
        return View(product);
    }

    // UPDATE - Save changes
    [HttpPost]
    public IActionResult Edit(Product product)
    {
        var oldProduct = products.Find(p => p.Id == product.Id);

        if (oldProduct != null)
        {
            oldProduct.Name = product.Name;
            oldProduct.Price = product.Price;
        }

        return RedirectToAction("Index");
    }

    // DELETE
    public IActionResult Delete(int id)
    {
        var product = products.Find(p => p.Id == id);

        if (product != null)
        {
            products.Remove(product);
        }

        return RedirectToAction("Index");
    }
}
Now we have:
CRUD -> Code
Create -> Initial products in List
Read -> Index()
Update -> Edit()
Delete -> Delete()

using Microsoft.AspNetCore.Mvc;
using StudentAPI.Models;

namespace StudentAPI.Controllers;

[ApiController]
[Route("api/[controller]")]
public class StudentController : ControllerBase
{
    static List<Student> students = new List<Student>
    {
        new Student { Id = 1, Name = "Devesh" },
        new Student { Id = 2, Name = "Rahul" }
    };

    // GET
    [HttpGet]
    public IActionResult Get()
    {
        return Ok(students);
    }

    // POST
    [HttpPost]
    public IActionResult Add(Student student)
    {
        students.Add(student);
        return Ok(student);
    }

    // PUT
    [HttpPut("{id}")]
    public IActionResult Update(int id, Student student)
    {
        var oldStudent = students.Find(s => s.Id == id);

        if (oldStudent == null)
            return NotFound();

        oldStudent.Name = student.Name;

        return Ok(oldStudent);
    }

    // DELETE
    [HttpDelete("{id}")]
    public IActionResult Delete(int id)
    {
        var student = students.Find(s => s.Id == id);

        if (student == null)
            return NotFound();

        students.Remove(student);

        return Ok(student);
    }
}
