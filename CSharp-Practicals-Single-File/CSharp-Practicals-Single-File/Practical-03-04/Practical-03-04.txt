PRACTICAL 3 + 4 — SHOPPING CART CRUD

Use predefined products using List<Product>. No database.

Step 1: Create ASP.NET Core Web App (Model-View-Controller)
Project name: ShoppingCart
Authentication: None
HTTPS enabled.

Step 2: Models/Product.cs

namespace ShoppingCart.Models;

public class Product
{
    public int Id { get; set; }
    public string Name { get; set; } = "";
    public int Price { get; set; }
}

Step 3: Controllers/ProductController.cs

using Microsoft.AspNetCore.Mvc;
using ShoppingCart.Models;

namespace ShoppingCart.Controllers;

public class ProductController : Controller
{
    static List<Product> products = new List<Product>
    {
        new Product { Id = 1, Name = "Mouse", Price = 500 },
        new Product { Id = 2, Name = "Keyboard", Price = 1000 },
        new Product { Id = 3, Name = "Headphones", Price = 1500 }
    };

    public IActionResult Index()
    {
        return View(products);
    }

    public IActionResult Edit(int id)
    {
        var product = products.Find(p => p.Id == id);
        return View(product);
    }

    [HttpPost]
    public IActionResult Edit(Product product)
    {
        var oldProduct = products.Find(p => p.Id == product.Id);

        if (oldProduct != null)
        {
            oldProduct.Name = product.Name;
            oldProduct.Price = product.Price;
        }

        return RedirectToAction("Index");
    }

    public IActionResult Delete(int id)
    {
        var product = products.Find(p => p.Id == id);

        if (product != null)
        {
            products.Remove(product);
        }

        return RedirectToAction("Index");
    }
}
